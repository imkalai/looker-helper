views/orders.view.lkml · version 2
view: orders {
  sql_table_name: `acme-analytics.sales.orders` ;;

  dimension: order_id {
    primary_key: yes
    sql: ${TABLE}.order_id ;;
  }

  dimension_group: created {
    type: time
    timeframes: [date, week, month, quarter, year]
    sql: ${TABLE}.created_at ;;
  }

  measure: total_revenue {
    label: "Total revenue"
    type: sum
    sql: ${TABLE}.amount ;;
    value_format_name: gbp
  }

  measure: order_count {
    label: "Number of orders"
    type: count
  }

  measure: avg_order_value {
    label: "Average order value"
    type: number
    sql: ${total_revenue} / NULLIF(${order_count}, 0) ;;
    value_format_name: gbp
  }
}
view: customers {
  sql_table_name: `acme-analytics.sales.customers` ;;

  dimension: customer_id {
    primary_key: yes
    sql: ${TABLE}.customer_id ;;
  }

  dimension: region {
    label: "Customer region"
    sql: ${TABLE}.region ;;
  }
}
connection: "bq_acme_analytics"

include: "/views/*.view.lkml"
include: "/dashboards/*.dashboard.lookml"

explore: orders {
  sql_always_where: ${orders.status} != 'cancelled' ;;
  join: customers {
    type: left_outer
    relationship: many_to_one
    sql_on: ${orders.customer_id} = ${customers.customer_id} ;;
  }
}
- dashboard: monthly_sales
  title: Monthly Sales by Region
  layout: newspaper
  elements:

    model: sales
    explore: orders
    type: single_value

  - title: Revenue by month
    model: sales
    explore: orders

  - title: Revenue by region
    model: sales
    explore: orders
    type: looker_bar
    fields: [customers.region, orders.total_revenue]
